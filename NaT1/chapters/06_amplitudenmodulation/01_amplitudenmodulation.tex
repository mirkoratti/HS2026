\textbf{Mischer:} Basisband $\xrightarrow{\times\cos}$ Übertragungsband $\xrightarrow{\times\cos}$ Basisband bzw. Zwischenfrequenz $f_{ZF}$.
\textbf{Superhet:} ZF-Filter fest, LO verstellbar: $|f_{LO}-f_c| = f_{ZF}$ (MW: $f_{ZF}=\qty{455}{kHz}$).\\
\textbf{Frequenzmultiplex (FDM):} verschiedene Träger $\omega_{c,i}$, Bänder dürfen nicht überlappen: $|\omega_{c,i+1}-\omega_{c,i}| \geq 2\pi

{\color{accent}} l'offset \glqq$1+$\grqq{} tiene l'inviluppo $A_c(1+\mu m_n)$ sempre positivo $\Rightarrow$ basta un \textbf{Hüllkurvendetektor} (ricevitore più semplice: radio AM, oggi Flugfunk).

\paragraph{Äquivalente Basisbanddarstellung S. 53}
{\color{accent}\textit{Idea:}} si sostituisce la portante reale ad alta frequenza con un \textbf{komplexer Zeiger} $x_{eq}(t)$ che contiene solo ampiezza e fase $\rightarrow$ si calcola solo col segnale lento in banda base (solo per portanti reali sinusoidali).

\begin{minipage}[t]{.52\textwidth}
	$x_{AM}(t) = A_c(1+\mu m_n)\cdot\tfrac{1}{2}\left(e^{+j\omega_c t}+e^{-j\omega_c t}\right)$\\[.05cm]
	\emph{Tre semplificazioni:}
	\begin{enumerate}\setlength{\itemsep}{0pt}\setlength{\parsep}{0pt}\setlength{\topsep}{0pt}
		\item \textbf{togli il Zeiger a $-\omega_c$}: è sempre il coniugato, nessuna info in più ($v+v^*=\Re\{2v\}$)
		\item \textbf{non scrivere $\Re\{\cdot\}$}: sottinteso, si prende alla fine
		\item \textbf{non scrivere $e^{+j\omega_c t}$}: rotazione nota e uguale per tutti, non cambia la \textbf{Umhüllende}
	\end{enumerate}
\end{minipage}\hfill
\begin{minipage}[t]{.45\textwidth}
	\efbox[linecolor=red]{$x(t) = \Re\{x_{eq}(t)\, e^{j\omega_c t}\}$, \; $x_{eq} = I + jQ = |x_{eq}|\, e^{j\varphi}$}\\[.05cm]
	$\Rightarrow x(t) = I\cos(\omega_c t) - Q\sin(\omega_c t)$\\
	$I$: Inphase (asse orizz.), $Q$: Quadratur (asse vert.)\\
	$|x_{eq}|$ = Umhüllende, $\arg x_{eq}$ = Phase\\[.05cm]
	\textbf{Träger $\rightarrow$ Zeiger:} $\cos(\omega_c t+\varphi)\rightarrow e^{j\varphi}$\\
	$\cos \rightarrow 1,\ \sin \rightarrow -j,\ -\cos \rightarrow -1,\ -\sin \rightarrow +j$
\end{minipage}

\vspace{.1cm}
\textbf{Ortskurve} di $x_{eq}$ = tutti i valori assunti (senza l'andamento nel tempo). AM: il punto si muove solo su una \textbf{retta per l'origine} (fase fissa, cambia solo la lunghezza).

\subsubsection{Quadraturamplitudenmodulation (QAM) S. 64}
\efbox[linecolor=red]{$x_{QAM}(t) = A_c\, m_1(t) \cos(\omega_c t) + A_c\, m_2(t) \sin(\omega_c t)$} = zwei DSB-SC im selben Band, orthogonale Träger\\
{\color{accent}} $\cos$ e $\sin$ sono \textbf{orthogonal}: i due messaggi si sovrappongono in frequenza ma in ricezione si separano, perché ogni ramo vede l'altro solo a $2\omega_c$.\\
Vorteil: Bandbreiteneffizienz wie SSB $\cdot$ Nachteil: Kohärenz muss sehr präzis sein (Trainingssequenz in Präambel) $\cdot$ Bsp.: Modems V.29/V.32, OFDM-Subträger, Chrominanz PAL/NTSC

\includegraphics[width=.75\linewidth]{chapters/06_amplitudenmodulation/figures/abbildung5_25.pdf}

\begin{minipage}[t]{.46\linewidth}\raggedright
	\textbf{Modulator} = Hartley-Struktur, aber $m_1$, $m_2$ unabhängig\\
	\efbox[linecolor=red]{$B_{QAM} = 2\cdot \max(B_{m1}, B_{m2})$} $\rightarrow$ wie DSB\\
	$m_1$ = Hilberttrafo von $m_2$ $\Rightarrow$ $B_{SSB} = B_m$\\
	Teilspektren addieren sich komplex, aber: \efbox[linecolor=red]{$P_{QAM} = P_{DSB1} + P_{DSB2}$}

	\textbf{Kohärente Demodulation} (Zweig $\cos$):\\
	$A_c m_1 \cos\cdot\cos = \frac{A_c}{2} m_1 [1 + \cos(2\omega_c t)] \xrightarrow{TP} \frac{A_c}{2} m_1$\\
	$A_c m_2 \sin\cdot\cos = \frac{A_c}{2} m_2 \sin(2\omega_c t) \xrightarrow{TP} 0$\\
	Zweig $\sin$ analog $\Rightarrow$ $y_{m2} = \frac{A_c}{2} m_2(t)$
