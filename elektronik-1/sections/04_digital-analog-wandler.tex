\section{Digital-Analog-Wandler (DAC)}\label{sec:dac}

\subsection{Grundbegriffe}\label{sec:dac-grund}

$$ \boxed{ q = 1\,{\rm LSB} = \frac{V_{\rm refp}-V_{\rm refn}}{2^n} } \qquad
   \boxed{ V_{\rm out} = V_{\rm refn} + D\cdot q } \qquad D = 0\dots 2^n-1 $$
Höchster Wert: \; $V_{\rm out,max} = V_{\rm refp} - q$ \; (ein LSB unter der Referenz)

\vspace{2pt}
{\renewcommand{\arraystretch}{1.45}
\begin{tabularx}{\columnwidth}{@{}l l X@{}}
	\toprule
	\textbf{Prinzip}             & \textbf{Elemente}             & \textbf{Eigenschaft} \\ \midrule
	Parallel (String)            & $2^n$ Widerstände und Schalter & garantiert monoton \\
	Wäge (gewichtet, R-2R, $C$)  & $n$ Elemente                  & nicht garantiert monoton \\
	Zähl (PWM, Sigma-Delta)      & 1 Schalter + $RC$             & sehr einfach, aber langsam \\ \bottomrule
\end{tabularx}}
\textrightarrow\ Parallel- und Wägeverfahren werden oft kombiniert.

\end{tabularx}}

\textbf{Varianten}: Strom-DAC parallel ($I_{\rm out} = D\cdot I$, Ströme nach $I_{\rm out}$ oder $\overline{I_{\rm out}}$); \;
ohne Decoder mit zusätzlichen Schaltern (1-Bit-DACs in Serie).

\warn{\textit{Errore dovuto al carico: zero a $D=0$, il più grande a metà scala.}}

\subsection{Wägeverfahren}\label{sec:dac-waege}

{\renewcommand{\arraystretch}{1.75}
\begin{tabularx}{\columnwidth}{@{}l l X@{}}
	\toprule
	\textbf{Variante}        & \textbf{$V_{\rm out}$}                    & \textbf{Merkmal} \\ \midrule
	Gewichtete $R$, Teiler   & $V_{\rm refn} + D\cdot q$
	                         & Leitwerte $2^i G_0$ addieren; \; $R_{\rm th} = \dfrac{1}{2^n G_0}$ \\
	Gewichtete $R$ + OpAmp   & $V_{\rm ref0} - (V_{\rm ref1}-V_{\rm ref0})\,R_{fb}\displaystyle\sum_i \frac{B_i}{R_i}$
	                         & $q = \dfrac{R_{fb}}{R_{\rm LSB}}\,|\Delta V_{\rm ref}|$, \; belastbar \\
	R-2R + OpAmp             & $V_{\rm refn} - (V_{\rm refp}-V_{\rm refn})\,\dfrac{D}{2^n}$
	                         & nur $R$ und $2R$, jeder Knoten halbiert \\
	Kapazitiv mit $C_{fb}$   & $V_{\rm ref0} - (V_{\rm ref1}-V_{\rm ref0})\,\dfrac{C_{\rm tot}}{C_{fb}}$
	                         & PH0: Reset, PH1: gültig; S\&H nötig \\
	Multiplizierend          & wie R-2R, $V_{\rm ref}$ = Eingangssignal
	                         & auch AC und negativ: 2 Quadranten \\ \bottomrule
\end{tabularx}}

\begin{rezept}{Rezept: R-2R ohne Ersatzwiderstand}
	\begin{steps}
		\item Von rechts schauen: jeder Knoten sieht $R_{\rm th} = R$
		\item Das MSB trägt $\frac{\Delta V}{2}$ bei, das nächste Bit $\frac{\Delta V}{4}$, dann $\frac{\Delta V}{8}$ \dots
		\item Superposition: Beiträge der geschalteten Bits addieren (beim OpAmp mit Vorzeichen $-$)
	\end{steps}
\end{rezept}

\begin{warnbox}
	\small
	\warn{\textit{R-2R e capacitivo con OpAmp invertono: segno meno!}}
	\warn{\textit{$R_{\rm LSB}$ è la R \textbf{più grande} (passa la corrente più piccola).}}
	\warn{\textit{Capacitivo: $C_{\rm tot}$ = somma dei C collegati a $V_{\rm ref1}$ (carica conservata, vedi Anhang).}}
\end{warnbox}

\subsection{Fehler und Toleranzen}\label{sec:dac-fehler}

{\renewcommand{\arraystretch}{1.45}
\begin{tabularx}{\columnwidth}{@{}l X@{}}
	\toprule
	\textbf{Fehler} & \textbf{Bedeutung} \\ \midrule
	Offset          & ganze Kennlinie verschoben \\
	Gain            & nach Offsetabgleich: Fehler beim maximalen Code \\
	INL             & grösste Abweichung Ist--Soll (nach Offset- und Gainabgleich) \\
	DNL             & Abweichung eines Sprungs von 1 LSB; \; ${\rm DNL} < -1$ \textrightarrow\ nicht monoton \\
	Settling Time   & Zeit, bis $V_{\rm out}$ im Band $\pm\frac12 q$ bleibt \\ \bottomrule
\end{tabularx}}

\textbf{Worst-Case} (Rezept \ref{sec:worstcase}), String-DAC als Teiler:
$$ V_{\min} = \Delta V\,\frac{R_{\rm unten}(1-t)}{R_{\rm unten}(1-t) + R_{\rm oben}(1+t)} \qquad V_{\max}\text{: Vorzeichen umgekehrt} $$
\textbf{Kritischster Codewechsel} am MSB ($2^{n-1}-1 \to 2^{n-1}$): alle Bits schalten gleichzeitig \textrightarrow\ Glitch, Monotonie gefährdet.

\warn{\textit{Offset dell'OpAmp nel DAC con OpAmp: offset \textbf{e} errore di gain, perché il $G_N$ cambia con il codice.}}

\vspace{2pt}
Zähler bis $N = 2^{\rm Bits}$, Code $n$, Takt $T_{\rm clk} = 1/f_{\rm clk}$; \; $RC$ glättet zu «DC».
$$ T = N\,T_{\rm clk} \qquad t_{\rm on} = n\,T_{\rm clk} \qquad
   \boxed{ \overline{V_{\rm out}} = V_{\rm refn} + (V_{\rm refp}-V_{\rm refn})\,\frac{n}{N} } $$

{\renewcommand{\arraystretch}{1.8}
\begin{tabularx}{\columnwidth}{@{}l X@{}}
	\toprule
	\textbf{Grösse}            & \textbf{Formel} \\ \midrule
	Rippel ($RC \gg T$)        & $V_{\rm rip} \approx \Delta V\,\dfrac{t_{\rm on}\,t_{\rm off}}{RC\;T}$, \;
	                             Maximum bei $n = \frac N2$: \; $\Delta V\,\dfrac{T}{4RC}$ \\
	Rippel exakt ($n=\frac N2$) & $V_{\rm rip} = \Delta V\,\dfrac{1-\e^{-T/2RC}}{1+\e^{-T/2RC}}$ \\
	Einschwingen               & $t = RC\,\ln\dfrac{|V_{\rm start}-V_{\rm end}|}{\text{Fehlerband}}$; \;
	                             voller Sprung auf $\pm\frac12$ LSB: \; $t = RC\,\ln 2^{\,{\rm Bits}+1}$ \\ \bottomrule
\end{tabularx}}

\textbf{Rippel kleiner}: grösseres $RC$, Filter höherer Ordnung, kürzere Periode, oder \textbf{Sigma-Delta}:
Akkumulator addiert $n$ bei jedem Takt, der Übertrag steuert den Schalter
\textrightarrow\ gleicher Mittelwert, die «1» verteilt \textrightarrow\ weniger Rippel, mehr Schaltvorgänge.

\warn{\textit{«Kleiner als 1 LSB» e «entro ½ LSB» sono bande diverse: guarda cosa chiede.}}

\subsection{Kombinierte DACs}

\begin{itemize}[nosep,leftmargin=*]
	\item \textbf{Kaskadiert}: MS-DAC liefert 2 Spannungen (über und unter dem Ziel), LS-DAC teilt dieses Intervall \textrightarrow\ kleineres $q$
	\item \textbf{Segmentiert}: hohe Bits als $2^k-1$ gleiche Elemente, Rest binär \textrightarrow\ kleiner Glitch am MSB
	\item \textbf{Schnelle DACs}: Stromausgang + OpAmp, denn Ströme schalten schneller als Spannungen
\end{itemize}
