\section{Arm Cortex-M4 \& Timer}

\subsection{Watchdog Window-Watchdog-Timer (WWDT)}
\begin{itemize}
    \item Zur Überwachung der Code-Ausführung. Erzeugt im Fehlerfall einen Reset.
    \item Dieser Timer muss jeweils vor Ablauf der Timeout-Zeit zurückgesetzt werden.
    \item Ansonsten wird eine definierte Aktion ausgeführt (z.B. Reset).
\end{itemize}

\end{tabularx}

\subsubsection{Timer: PWM messen (Input-Capture)}
\begin{itemize}
    \item \textbf{Ch1} (steigende Flanke, Reset-Capture): Counter wird gespeichert und auf $0$ gesetzt $\Rightarrow$ \textcolor{orange!80!black}{CCR1} = Periode.
    \item \textbf{Ch2} (fallende Flanke, Capture): \textcolor{blue!70!black}{CCR2} = Pulsdauer. Duty $\approx (\mathrm{CCR2}+1)/(\mathrm{CCR1}+1)$.
    % \item $\mathrm{CCR}=\dfrac{t}{T_{\rm tick}}-1$\quad $T_{\rm tick}=\dfrac{\mathrm{CKD}\,(\mathrm{PSC}+1)}{f_{in}}$ (CKD = Internal Clock Division: 1, 2, 4).
\end{itemize}

\subsection{M4 vs M0}
\begin{itemize}
    \item \textbf{M4:} High Performance / neuere Technologie, schnellere Taktung und Codeausführung (Thumb-2). Zusätzlich: FPU, erweiterte Debug-Unit, Harvard-Architektur, Bit-Banding, SIMD, DSP-Funktionen, ALU, Barrel Shifter, MAC.
    \item \textbf{M0:} Low Power / Low Cost (alte Technologie)
\end{itemize}

\subsubsection{Cortex-M4 Prozessor}

\begin{itemize}
        \item 3-stufige Pipeline für Thumb-2 Instruktionssatz
        \item NVIC, SYSTICK \& leistungsfähige debug Units
        \item Memory Protection Unit (MPU) und Floating Point Unit (FPU)
\end{itemize}

\begin{minipage}{0.5\columnwidth}
    \begin{itemize}
        \item Erweiterte Busstruktur
        \begin{description}
            \item[\cgn{I-Code Bus}] Instruction- \& Vector-Fetch; 32-Bit-Zugriff (Thumb-1).
            \item[\cbl{D-Code Bus}] Daten- \& Debugzugriff auf ROM (Konstanten); unaligned Zugriff möglich (wird in aligned gewandelt).
            \item[\crd{System Bus}] Zugriff auf SRAM \& Peripherie (Variablen); für Code \& Daten; unaligned Zugriff möglich.
            \item[\cvt{PP Bus}] Privater Peripheriebus (Debug, MPU, FPU); nur word-aligned Transfer.
        \end{description}
     
    \end{itemize}
\end{minipage}
